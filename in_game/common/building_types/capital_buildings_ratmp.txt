local_governor = {
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	important_for_UI = yes
	
	employment_size = large_noble_employment
	
	category = government_category
	
	rural_settlement = no
	town = no
	city = yes
	megalopolis = yes
		 
	allow = {
		owner = scope:actor #not in subjects
	}

	location_potential = {
		has_road_to_capital = yes
		owner.capital != this
		OR = {
			AND = {
				NOT = {
					region = {
						any_ownable_location_in_region = {
							count >= 1
							has_building = building_type:local_governor
							owner = scope:actor
						}
					}
				}
				region = {
					any_ownable_location_in_region = {
						is_owned_or_owned_by_subjects_or_below_of = scope:actor
						percent >= 0.25
					}
				}
			}
			AND = {
				NOT = {
					region = {
						any_ownable_location_in_region = {
							count >= 2
							has_building = building_type:local_governor
							owner = scope:actor
						}
					}
				}
				region = {
					any_ownable_location_in_region = {
						is_owned_or_owned_by_subjects_or_below_of = scope:actor
						percent >= 0.51
					}
				}
			}
		}

	}


	expensive = yes
	
	automation_build_allowed = no
	ai_forbid_shutdown = yes
	forbidden_for_estates = yes


	build_time = government_build_time


	unique_production_methods = {
		local_governor_maintenance = {
			tools = 1
			masonry = 1
			weaponry = 0.5
			paper = 3
			books = 1
			goods_gold = 1
			furniture = 2
			fine_cloth = 1

			category = building_maintenance
		}
	}


	construction_demand = local_governor_construction

	modifier = {
		local_proximity_source = 80
	}
}

relocate_population = {
	ability = adm
	
	allow_multiple = yes
	icon = pop_promote_action
	potential = {
		is_rebel_country = no
	}
	
	allow = {
		total_population >= 5
	}
	
	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		cache_targets = yes
		name = "send_people_to_the_colonies_source"
		none_available_msg_key = "send_people_to_the_colonies_source_no_provinces"
		column = {
			data = name
		}
		column = {
			data = population
		}
		enabled = {
			population > 1
		}
		map_color = define:NMapColors|MAP_COLOR_HIGH
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target_1
		name = "send_people_to_the_colonies_destination"
		none_available_msg_key = "send_people_to_the_colonies_destination_no_provinces"
		column = {
			data = name
		}
		column = {
			data = population
		}
		visible = {
			scope:target = { is_overseas_for_owner = no }
		}
		enabled = {
			population < scope:target.population 
		}
		map_color = define:NMapColors|MAP_COLOR_HIGH
	}

	on_activate = {
		if = {
			limit = {
				exists = scope:target
				exists = scope:target_1
			}
			add_migration = {
				owner = scope:actor
				from = scope:target.province_definition
				to = scope:target_1.province_definition
				amount = {
					value = 0.100
					multiply = {
						value = 1
						if = {
							limit = {
								exists = scope:cabinet
							}
							add = "scope:cabinet.effective_skill"
						}
					}
				}
				months = -1
			}
		}
	}

	on_deactivate = {
		if = {
			limit = {
				exists = scope:target
				exists = scope:target_1
			}
			remove_migration = {
				owner = scope:actor
				from = scope:target.province_definition
				to = scope:target_1.province_definition
			}
		}
	}

	country_modifier = {
	}

	is_finished = {
		trigger_if = {
			limit = {
				exists = scope:target
				exists = scope:target_1
			}
		}
	}
}
